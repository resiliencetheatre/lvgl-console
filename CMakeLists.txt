cmake_minimum_required(VERSION 3.18)
project(lvgl_console VERSION 0.1.0 LANGUAGES C CXX ASM)
include(GNUInstallDirs)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(PNG REQUIRED)
find_package(ALSA REQUIRED)
set(LVGL_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/lvgl" CACHE PATH "LVGL 9.6.0 source directory")
set(LVGL_CONSOLE_ASSET_DIR "${CMAKE_INSTALL_FULL_DATADIR}/lvgl-console" CACHE STRING "Runtime asset directory")
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/config/lv_conf.h" CACHE PATH "" FORCE)
set(LV_BUILD_USE_KCONFIG OFF CACHE BOOL "" FORCE)
set(LV_BUILD_SET_CONFIG_OPTS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/lvgl" EXCLUDE_FROM_ALL)
target_compile_definitions(lvgl PUBLIC APP_ASSET_DIR="${LVGL_CONSOLE_ASSET_DIR}")
target_link_libraries(lvgl PUBLIC Threads::Threads PNG::PNG m)
add_executable(lvgl-console src/main.c src/ui.c src/chat_backend.c src/network_status.c src/keyboard.c src/ping_status.c src/ptt_status.c src/settings.c src/volume.c)
target_link_libraries(lvgl-console PRIVATE lvgl::lvgl ALSA::ALSA)
target_compile_options(lvgl-console PRIVATE -Wall -Wextra -Wpedantic)
add_executable(lvgl-console-launcher src/launcher.c)
target_compile_definitions(lvgl-console-launcher PRIVATE
    APP_EXECUTABLE="${CMAKE_INSTALL_FULL_BINDIR}/lvgl-console")
target_compile_options(lvgl-console-launcher PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS lvgl-console lvgl-console-launcher RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(PROGRAMS tools/udpptt-audio-select DESTINATION ${CMAKE_INSTALL_LIBEXECDIR})
install(FILES config/udpptt.usb-hf config/udpptt.usb-monofone DESTINATION /etc/default)
install(FILES assets/link.png DESTINATION ${CMAKE_INSTALL_DATADIR}/lvgl-console)

option(LVGL_CONSOLE_TESTS "Build headless UI smoke test" OFF)
